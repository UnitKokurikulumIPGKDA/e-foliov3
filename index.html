/** * Menu tersuai yang muncul secara automatik bila spreadsheet dibuka. * Membolehkan sesiapa dengan akses Editor menjalankan setup tanpa masuk ke Apps Script. */ function onOpen() { SpreadsheetApp.getUi() .createMenu('⚙️ Setup E-Folio') .addItem('Cipta / Semak Struktur Sheet', 'setupDatabaseSystem') .addItem('Cipta Akaun Admin Pertama', 'promptCreateFirstAdmin') .addToUi(); } /** * FUNGSI UTAMA: Cipta semua sheet & header yang diperlukan sistem secara automatik. * Selamat dijalankan berulang kali (idempotent) — tidak akan menimpa data sedia ada. */ function setupDatabaseSystem() { const ss = SpreadsheetApp.getActiveSpreadsheet(); } // ============================================ // FUNGSI DRIVE - SALIN FAIL KE FOLDER TERSUSUN // ============================================ /** * Dapatkan sub-folder mengikut nama; cipta jika belum wujud. */ function getOrCreateSubFolder(parentFolder, folderName) { const safeName = String(folderName || 'Umum').trim() || 'Umum'; const existing = parentFolder.getFoldersByName(safeName); if (existing.hasNext()) { return existing.next(); } return parentFolder.createFolder(safeName); } /** * Ekstrak File ID daripada pelbagai format pautan Google Drive. */ function extractDriveFileId(url) { if (!url) return null; const str = String(url); const patterns = [ /\/d\/([a-zA-Z0-9_-]{20,})/, /[?&]id=([a-zA-Z0-9_-]{20,})/ ]; for (const p of patterns) { const match = str.match(p); if (match && match[1]) return match[1]; } return null; } /** * Salin fail Drive ke dalam struktur folder tersusun: * Root ▸ Pensyarah ▸ Tahun ▸ Kategori, dinamakan mengikut tajuk rekod. */ function copyFileToOrganizedFolder(pautanAsal, ownerUsername, tahun, kategori, namaRekod) { const fileId = extractDriveFileId(pautanAsal); if (!fileId) { return { success: false, url: pautanAsal, message: 'Bukan pautan fail Drive individu yang sah; pautan asal dikekalkan.' }; } try { const rootFolder = DriveApp.getFolderById(ROOT_FOLDER_ID); const ownerFolder = getOrCreateSubFolder(rootFolder, ownerUsername); const yearFolder = getOrCreateSubFolder(ownerFolder, tahun); const catFolder = getOrCreateSubFolder(yearFolder, kategori); let sourceFile; try { sourceFile = DriveApp.getFileById(fileId); } catch (accessErr) { return { success: false, url: pautanAsal, message: 'Skrip tiada akses kepada fail ini; pautan asal dikekalkan.' }; } const cleanName = String(namaRekod || sourceFile.getName()).trim().substring(0, 150) || sourceFile.getName(); const copiedFile = sourceFile.makeCopy(cleanName, catFolder); copiedFile.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return { success: true, url: copiedFile.getUrl() }; } catch (err) { return { success: false, url: pautanAsal, message: 'Ralat semasa menyalin fail: ' + err.toString() }; } } // ============================================ // doGet - PENGENDALI PERMINTAAN HTTP GET // ============================================ /** * Mengambil data daripada lembaran yang dinyatakan * (Users, LecturerList, Data_Folio, Folio_Kursus, Subjek_Diajar) * @param {Object} e - Objek event daripada pelayan * @returns {TextOutput} JSON response */ function doGet(e) { const lock = LockService.getScriptLock(); lock.tryLock(10000); try { const sheetName = (e && e.parameter && e.parameter.sheet) ? String(e.parameter.sheet).trim() : 'Data_Folio'; const rawUsername = (e && e.parameter && e.parameter.username) ? String(e.parameter.username).trim().toLowerCase() : null; const role = (e && e.parameter && e.parameter.role) ? String(e.parameter.role).trim() : null; const viewUser = (e && e.parameter && e.parameter.viewUser) ? String(e.parameter.viewUser).trim().toLowerCase() : null; const ss = SpreadsheetApp.openById(SPREADSHEET_ID); if (sheetName === 'LecturerList') { const userSheet = ss.getSheetByName('Users'); if (!userSheet) return createJsonResponse([]); const data = getSheetDataAsJson(userSheet); const lecturers = data.map(u => ({ Username: String(u.Username || '').trim(), Nama_Penuh: u.Nama_Penuh || u.Username, No_Pekerja: u.No_Pekerja || '', Gred: u.Gred || '', Jabatan: u.Jabatan || '', Role: u.Role || 'Pensyarah', Foto_URL: u.Foto_URL || '' })); return createJsonResponse(lecturers); } const sheet = ss.getSheetByName(sheetName); if (!sheet) { return createJsonResponse([]); } let records = getSheetDataAsJson(sheet); if (sheetName !== 'Users' && rawUsername) { const hasUsernameColumn = records.length > 0 && ('Username' in records[0]); const isPrivileged = (role === 'Admin') || (rawUsername === 'auditor') || rawUsername.includes('admin'); if (isPrivileged) { if (viewUser && viewUser !== 'semua' && hasUsernameColumn) { records = records.filter(r => { const rowOwner = String(r.Username || r.Nama || '').trim().toLowerCase(); return rowOwner === viewUser; }); } } else if (hasUsernameColumn) { records = records.filter(r => { const rowOwner = String(r.Username || '').trim().toLowerCase(); return !rowOwner || rowOwner === rawUsername; }); } } return createJsonResponse(records); } catch (error) { return createJsonResponse({ result: "error", message: error.toString() }); } finally { lock.releaseLock(); } } // ============================================ // doPost - PENGENDALI PERMINTAAN HTTP POST // ============================================ /** * Mengendalikan log masuk Google, pendaftaran automatik, kemaskini profil, dan CRUD rekod * @param {Object} e - Objek event mengandungi body POST JSON * @returns {TextOutput} JSON response */ function doPost(e) { const lock = LockService.getScriptLock(); lock.tryLock(10000); try { if (!e || !e.postData || !e.postData.contents) { return createJsonResponse({ result: 'error', message: 'Payload JSON tidak diterima.' }); } const rawData = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.openById(SPREADSHEET_ID); const action = rawData.action; // A. LOG MASUK GOOGLE & PENDAFTARAN KENDIRI AUTOMATIK (TANPA KATA LALUAN) if (action === 'googleLogin' || action === 'login') { let userSheet = ss.getSheetByName('Users'); if (!userSheet) { userSheet = ss.insertSheet('Users'); userSheet.appendRow(['Username', 'Password', 'Nama_Penuh', 'No_Pekerja', 'Gred', 'Jabatan', 'Role', 'Foto_URL']); } const inputEmail = String(rawData.email || rawData.username || '').trim().toLowerCase(); const inputName = String(rawData.name || rawData.nama || '').trim(); const inputPicture = String(rawData.picture || rawData.foto || '').trim(); if (!inputEmail) { return createJsonResponse({ result: 'failed', message: 'Sila masukkan alamat emel Google yang sah.' }); } const users = getSheetDataAsJson(userSheet); let matchedUser = users.find(u => String(u.Username || '').trim().toLowerCase() === inputEmail); if (!matchedUser) { const isAdminAccount = inputEmail.includes('admin') || inputEmail === 'auditor' || inputEmail.includes('haniff.faizal'); const defaultRole = isAdminAccount ? 'Admin' : 'Pensyarah'; const formattedName = inputName || inputEmail.split('@')[0].toUpperCase().replace(/\./g, ' '); const defaultPhoto = inputPicture || 'https://i.imgur.com/9uSlDpC.png'; const newRow = [ inputEmail, 'GOOGLE_AUTH', formattedName, '-', 'DG41 / DG44', 'JABATAN PENDIDIKAN', defaultRole, defaultPhoto ]; userSheet.appendRow(newRow); matchedUser = { Username: inputEmail, Nama_Penuh: formattedName, No_Pekerja: '-', Gred: 'DG41 / DG44', Jabatan: 'JABATAN PENDIDIKAN', Role: defaultRole, Foto_URL: defaultPhoto, isNewRegistration: true }; } else { if (inputPicture && (!matchedUser.Foto_URL || matchedUser.Foto_URL.includes('default'))) { matchedUser.Foto_URL = inputPicture; const headers = userSheet.getRange(1, 1, 1, userSheet.getLastColumn()).getValues()[0]; const fotoColIndex = headers.findIndex(h => String(h).trim().toLowerCase() === 'foto_url') + 1; if (fotoColIndex > 0 && matchedUser.rowIndex) { userSheet.getRange(matchedUser.rowIndex, fotoColIndex).setValue(inputPicture); } } } delete matchedUser.Password; return createJsonResponse({ result: 'success', user: matchedUser }); } // B. PENDAFTARAN KENDIRI PENSYARAH BAHARU if (action === 'selfRegister' || action === 'register') { let userSheet = ss.getSheetByName('Users'); if (!userSheet) { userSheet = ss.insertSheet('Users'); userSheet.appendRow(['Username', 'Password', 'Nama_Penuh', 'No_Pekerja', 'Gred', 'Jabatan', 'Role', 'Foto_URL']); } const inputEmail = String(rawData.newUsername || rawData.email || rawData.username || '').trim().toLowerCase(); const inputNama = String(rawData.newNamaPenuh || rawData.nama || rawData.nama_penuh || '').trim().toUpperCase(); const inputNoPekerja = String(rawData.newNoPekerja || rawData.no_pekerja || '-').trim(); const inputGred = String(rawData.newGred || rawData.gred || 'DG41').trim().toUpperCase(); const inputJabatan = String(rawData.newJabatan || rawData.jabatan || 'UNIT AKADEMIK').trim().toUpperCase(); const inputFoto = String(rawData.newFoto || rawData.foto_url || 'https://i.imgur.com/9uSlDpC.png').trim(); if (!inputEmail || !inputNama) { return createJsonResponse({ result: 'failed', message: 'Sila lengkapkan nama penuh dan alamat emel.' }); } const users = getSheetDataAsJson(userSheet); const existingUser = users.find(u => String(u.Username || '').trim().toLowerCase() === inputEmail); if (existingUser) { return createJsonResponse({ result: 'failed', message: 'Emel ini sudah didaftarkan. Sila gunakan tab Log Masuk.' }); } const newRow = [inputEmail, 'GOOGLE_AUTH', inputNama, inputNoPekerja, inputGred, inputJabatan, 'Pensyarah', inputFoto]; userSheet.appendRow(newRow); const registeredUser = { Username: inputEmail, Nama_Penuh: inputNama, No_Pekerja: inputNoPekerja, Gred: inputGred, Jabatan: inputJabatan, Role: 'Pensyarah', Foto_URL: inputFoto }; return createJsonResponse({ result: 'success', message: 'Pendaftaran berjaya!', user: registeredUser }); } // C. KEMASKINI PROFIL KENDIRI PENSYARAH if (action === 'updateProfile') { const userSheet = ss.getSheetByName('Users'); if (!userSheet) throw new Error("Lembaran 'Users' tidak wujud."); const targetEmail = String(rawData.username || '').trim().toLowerCase(); const users = getSheetDataAsJson(userSheet); const foundUser = users.find(u => String(u.Username || '').trim().toLowerCase() === targetEmail); if (!foundUser) { return createJsonResponse({ result: 'failed', message: 'Akaun pengguna tidak dijumpai.' }); } const rowIndex = foundUser.rowIndex; if (rawData.nama_penuh) userSheet.getRange(rowIndex, 3).setValue(String(rawData.nama_penuh).trim().toUpperCase()); if (rawData.no_pekerja) userSheet.getRange(rowIndex, 4).setValue(String(rawData.no_pekerja).trim()); if (rawData.gred) userSheet.getRange(rowIndex, 5).setValue(String(rawData.gred).trim().toUpperCase()); if (rawData.jabatan) userSheet.getRange(rowIndex, 6).setValue(String(rawData.jabatan).trim().toUpperCase()); if (rawData.foto_url) userSheet.getRange(rowIndex, 8).setValue(String(rawData.foto_url).trim()); return createJsonResponse({ result: 'success', message: 'Profil berjaya dikemaskini.' }); } // D. TAMBAH PENGGUNA OLEH ADMIN if (action === 'addUser') { if (String(rawData.role).toLowerCase() !== 'admin') { return createJsonResponse({ result: "error", message: "Hanya pentadbir boleh menggunakan fungsi ini." }); } let userSheet = ss.getSheetByName('Users'); if (!userSheet) { userSheet = ss.insertSheet('Users'); userSheet.appendRow(['Username', 'Password', 'Nama_Penuh', 'No_Pekerja', 'Gred', 'Jabatan', 'Role', 'Foto_URL']); } userSheet.appendRow([ String(rawData.newUsername).trim().toLowerCase(), String(rawData.newPassword || '123456'), String(rawData.newNamaPenuh).trim().toUpperCase(), String(rawData.newNoPekerja).trim(), String(rawData.newGred).trim().toUpperCase(), String(rawData.newJabatan).trim().toUpperCase(), String(rawData.newRole || 'Pensyarah'), String(rawData.newFoto || 'https://i.imgur.com/9uSlDpC.png') ]); return createJsonResponse({ result: 'success', message: 'Pensyarah baharu berjaya didaftarkan.' }); } // E. PENGURUSAN REKOD FOLIO (ADD / EDIT / DELETE) const targetSheetName = rawData.targetSheet || 'Data_Folio'; const targetSheet = ss.getSheetByName(targetSheetName); if (!targetSheet) throw new Error(`Lembaran '${targetSheetName}' tidak wujud.`); const headers = targetSheet.getRange(1, 1, 1, targetSheet.getLastColumn()).getValues()[0]; const targetUser = (rawData.role === 'Admin' && rawData.actingAsUser) ? rawData.actingAsUser : rawData.username; if (action === 'add') { let copyResult = { success: true, url: rawData.pautan }; if (rawData.pautan) { const namaUntukFail = rawData.nama || rawData.nama_kursus || rawData.nama_subjek || rawData.keterangan; copyResult = copyFileToOrganizedFolder( rawData.pautan, targetUser, rawData.tahun, rawData.kategori || 'Subjek', namaUntukFail ); rawData.pautan = copyResult.url; } const newRow = buildRowFromPayload(headers, rawData, targetUser); targetSheet.appendRow(newRow); return createJsonResponse({ result: 'success', message: copyResult.success ? 'Rekod berjaya disimpan dan bahan disalin ke Drive terpusat.' : ('Rekod disimpan. ' + copyResult.message) }); } if (action === 'edit') { const rowIndex = parseInt(rawData.rowIndex, 10); if (!rowIndex || rowIndex < 2) throw new Error("Indeks baris tidak sah."); const pautanColIndex = headers.findIndex(h => String(h).trim().toLowerCase() === 'pautan') + 1; const pautanLama = pautanColIndex > 0 ? String(targetSheet.getRange(rowIndex, pautanColIndex).getValue()) : ''; let copyResult = { success: true, url: rawData.pautan }; const pautanBerubah = rawData.pautan && rawData.pautan.trim() !== pautanLama.trim(); if (pautanBerubah) { const namaUntukFail = rawData.nama || rawData.nama_kursus || rawData.nama_subjek || rawData.keterangan; copyResult = copyFileToOrganizedFolder( rawData.pautan, targetUser, rawData.tahun, rawData.kategori || 'Subjek', namaUntukFail ); rawData.pautan = copyResult.url; } const updatedRow = buildRowFromPayload(headers, rawData, targetUser); targetSheet.getRange(rowIndex, 1, 1, updatedRow.length).setValues([updatedRow]); return createJsonResponse({ result: 'success', message: copyResult.success ? ' + copyResult.message) }); } if (action === 'delete') { const rowIndex = parseInt(rawData.rowIndex, 10); if (!rowIndex || rowIndex < 2) throw new Error("Indeks baris tidak sah."); targetSheet.deleteRow(rowIndex); return createJsonResponse({ result: 'success', message: 'Rekod berjaya dipadam.' }); } return createJsonResponse({ result: 'error', message: 'Tindakan tidak sah.' }); } catch (err) { return createJsonResponse({ result: 'error', message: err.toString() }); } finally { lock.releaseLock(); } } /** * Membina susunan baris nilai mengikut susunan lajur header pangkalan data sebenar */ function buildRowFromPayload(headers, data, ownerUsername) { return headers.map(header => { const key = header.toString().trim().toLowerCase(); switch (key) { case 'username': return ownerUsername || ''; case 'tahun': return data.tahun || ''; case 'semester': return data.semester || ''; case 'kategori': return data.kategori || ''; case 'nama': return data.nama || data.nama_kursus || data.nama_subjek || ''; case 'nama_kursus': return data.nama_kursus || data.nama || ''; case 'kod_subjek': return data.kod_subjek || ''; case 'nama_subjek': return data.nama_subjek || ''; case 'kelas': return data.kelas || ''; case 'pautan': return data.pautan || ''; case 'keterangan': return data.keterangan || data.komen || ''; case 'komen': return data.keterangan || data.komen || '-'; default: return ''; } }); } /** * Mengambil keseluruhan data lembaran dan menukarnya menjadi objek JSON */ function getSheetDataAsJson(sheet) { const lastRow = sheet.getLastRow(); const lastCol = sheet.getLastColumn(); if (lastRow < 2 || lastCol < 1) return []; const headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(h => String(h).trim()); const rows = sheet.getRange(2, 1, lastRow - 1, lastCol).getValues(); return rows.map((row, idx) => { const item = { rowIndex: idx + 2 }; headers.forEach((h, colIdx) => { item[h] = row[colIdx]; }); return item; }); } /** * Membina output HTTP JSON dengan Cross-Origin headers */ function createJsonResponse(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON);
